# Runtime dependencies for the container image — deliberately NOT requirements.txt.
#
# requirements.txt is the development environment: it carries shap, prophet,
# matplotlib, seaborn, mlflow, the SageMaker SDK and pytest. None of that is
# needed to answer a prediction request, and every megabyte of it would be
# pulled into the image, slowing down the cold start of every Lambda
# invocation for the rest of the project's life.
#
# Versions are IDENTICAL to requirements.txt wherever they overlap. This is not
# tidiness: the model artifact is a pickle, and unpickling a scikit-learn
# estimator under a different version than it was written with is a documented
# source of silently wrong predictions. The pin is the guarantee that the model
# meets the same library in production that it met in training.
#
# boto3 ships inside the Lambda runtime, so it is not installed here.

fastapi==0.115.6
